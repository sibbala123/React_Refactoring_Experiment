+import { useMemo } from 'react'
+
+function filterItems(items, query, predicate) {
+  const normalizedQuery = query.trim().toLowerCase()
+  return items.filter((item) => {
+    const matchesQuery =
+      item.title.toLowerCase().includes(normalizedQuery) ||
+      item.owner.toLowerCase().includes(normalizedQuery) ||
+      item.status.toLowerCase().includes(normalizedQuery)
+    const matchesPredicate = predicate === 'All' ? true : item.status === predicate
+    return matchesQuery && matchesPredicate
+  })
+}
+
+export default function useFilter(items, query, predicate) {
+  return useMemo(() => filterItems(items, query, predicate), [items, query, predicate])
+}
