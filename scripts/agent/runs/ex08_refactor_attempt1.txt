import { badgeColor } from '../utils/format'

const DEFAULT_BUTTON_STYLE = {
  padding: '6px 10px',
  borderRadius: 8,
  border: '1px solid #99a6b3',
}

const DEFAULT_LINE_STYLE = { margin: '4px 0' }

export default function ReportRowDetails({ report, ui, actions, ...legacy }) {
  const resolvedReport = report ?? legacy.report ?? legacy
  const resolvedUi = ui ?? legacy.ui ?? legacy
  const resolvedActions = actions ?? legacy.actions ?? legacy
  const hasDetails =
    resolvedReport &&
    (resolvedReport.title ||
      resolvedReport.owner ||
      resolvedReport.status ||
      resolvedReport.updatedDaysAgo !== undefined)

  if (!hasDetails && !legacy.children) {
    return null
  }

  const isLight = resolvedUi.isLight ?? (resolvedUi.theme ?? legacy.theme) === 'light'
  const containerStyle = {
    border: `1px solid ${isLight ? '#c8d1db' : '#4b5b6d'}`,
    borderRadius: 10,
    padding: 12,
    background: isLight ? '#ffffff' : '#2b3541',
    ...resolvedUi.containerStyle,
  }

  const heading = resolvedUi.heading ?? legacy.heading
  const headingStyle = { marginTop: 0, ...resolvedUi.headingStyle }
  const lineStyle = { ...DEFAULT_LINE_STYLE, ...resolvedUi.lineStyle }
  const labels = resolvedUi.labels ?? legacy.labels ?? {}

  const titleLabel = labels.title ?? legacy.titleLabel ?? 'Title'
  const ownerLabel = labels.owner ?? legacy.ownerLabel ?? 'Owner'
  const statusLabel = labels.status ?? legacy.statusLabel ?? 'Status'
  const updatedLabel = labels.updated ?? legacy.updatedLabel ?? 'Updated'

  const statusColor = resolvedUi.statusColor ?? legacy.statusColor
  const colorStatus = resolvedUi.colorStatus ?? legacy.colorStatus ?? statusColor !== undefined && statusColor !== null
  const statusStyle = colorStatus ? { color: statusColor ?? badgeColor(resolvedReport.status) } : null

  const actionButtons =
    resolvedActions.buttons ??
    resolvedActions.items ??
    legacy.buttons ??
    legacy.items ??
    [
      {
        key: 'open',
        handler: resolvedActions.onOpen ?? legacy.onOpen,
        label: resolvedActions.openLabel ?? legacy.openLabel ?? 'Open',
      },
      {
        key: 'close',
        handler: resolvedActions.onClose ?? legacy.onClose,
        label: resolvedActions.closeLabel ?? legacy.closeLabel ?? 'Close',
      },
      {
        key: 'pin',
        handler: resolvedActions.onPin ?? legacy.onPin,
        label: resolvedActions.pinLabel ?? legacy.pinLabel ?? 'Pin',
      },
      {
        key: 'archive',
        handler: resolvedActions.onArchive ?? legacy.onArchive,
        label: resolvedActions.archiveLabel ?? legacy.archiveLabel ?? 'Archive',
      },
    ]

  const visibleActions = (actionButtons || [])
    .map((action, index) => {
      if (!action || !action.handler) {
        return null
      }
      const label = action.label ?? action.text
      if (!label) {
        return null
      }
      return {
        key: action.key ?? label ?? index,
        label,
        handler: action.handler,
        style: action.style,
      }
    })
    .filter(Boolean)

  const buttonStyle = { ...DEFAULT_BUTTON_STYLE, ...resolvedUi.buttonStyle }

  return (
    <div style={containerStyle}>
      {heading && <h3 style={headingStyle}>{heading}</h3>}
      {resolvedReport.title && (
        <p style={lineStyle}>
          {titleLabel}: {resolvedReport.title}
        </p>
      )}
      {resolvedReport.owner && (
        <p style={lineStyle}>
          {ownerLabel}: {resolvedReport.owner}
        </p>
      )}
      {resolvedReport.status && (
        <p style={lineStyle}>
          {statusLabel}:{' '}
          {statusStyle ? <span style={statusStyle}>{resolvedReport.status}</span> : resolvedReport.status}
        </p>
      )}
      {resolvedReport.updatedDaysAgo !== undefined && resolvedReport.updatedDaysAgo !== null && (
        <p style={lineStyle}>
          {updatedLabel}: {resolvedReport.updatedDaysAgo} days ago
        </p>
      )}
      {legacy.children}
      {visibleActions.length === 1 && (
        <button
          onClick={visibleActions[0].handler}
          style={{ ...buttonStyle, ...visibleActions[0].style }}
        >
          {visibleActions[0].label}
        </button>
      )}
      {visibleActions.length > 1 && (
        <div style={{ display: 'flex', gap: 8, flexWrap: 'wrap', ...resolvedUi.actionsContainerStyle }}>
          {visibleActions.map((action) => (
            <button key={action.key} onClick={action.handler} style={{ ...buttonStyle, ...action.style }}>
              {action.label}
            </button>
          ))}
        </div>
      )}
    </div>
  )
}

import { useCallback, useEffect, useMemo, useState } from 'react'
import ReportRow from '../components/ReportRow'
import ReportRowDetails from '../components/ReportRowDetails'
import Toolbar from '../components/Toolbar'
import { REPORTS } from '../data/mockData'
import { badgeColor } from '../utils/format'

function filterReports(reports, query, status) {
  const normalizedQuery = query.trim().toLowerCase()
  return reports.filter((r) => {
    const matchesQuery =
      r.title.toLowerCase().includes(normalizedQuery) ||
      r.owner.toLowerCase().includes(normalizedQuery) ||
      r.status.toLowerCase().includes(normalizedQuery)
    const matchesStatus = status === 'All' ? true : r.status === status
    return matchesQuery && matchesStatus
  })
}

export default function ReportsPage({ theme, user }) {
  const isLight = theme === 'light'
  const [query, setQuery] = useState('')
  const [status, setStatus] = useState('All')
  const [opened, setOpened] = useState(null)
  const [toast, setToast] = useState('')
  const [counter, setCounter] = useState(0)
  const [showHelp, setShowHelp] = useState(false)
  const [pinReady, setPinReady] = useState(false)

  const filtered = useMemo(() => filterReports(REPORTS, query, status), [query, status])
  const readyCount = useMemo(() => REPORTS.filter((r) => r.status === 'Ready').length, [])
  const draftCount = useMemo(() => REPORTS.filter((r) => r.status === 'Draft').length, [])
  const blockedCount = useMemo(() => REPORTS.filter((r) => r.status === 'Blocked').length, [])

  useEffect(() => {
    if (!toast) {
      return undefined
    }
    const id = window.setTimeout(() => setToast(''), 1600)
    return () => window.clearTimeout(id)
  }, [toast])

  const openReport = (report) => {
    setOpened(report)
    setCounter((prev) => prev + 1)
    setToast(`Opened report: ${report.title}`)
  }

  const handleCloseReport = useCallback(() => setOpened(null), [setOpened])

  const openDetails = useMemo(
    () => ({
      report: opened,
      ui: {
        isLight,
        heading: 'Open Report',
        containerStyle: { marginTop: 12 },
        colorStatus: false,
      },
      actions: {
        onClose: handleCloseReport,
        closeLabel: 'Close',
      },
    }),
    [opened, isLight, handleCloseReport]
  )

  const pinnedReadyReports = pinReady ? REPORTS.filter((r) => r.status === 'Ready') : []

  return (
    <section>
      <Toolbar theme={theme} title="Reports" subtitle={`Status board for ${user.role}`} />

      <div
        style={{
          border: `1px solid ${isLight ? '#c8d1db' : '#4b5b6d'}`,
          borderRadius: 10,
          padding: 12,
          background: isLight ? '#ffffff' : '#2b3541',
          marginBottom: 12,
          display: 'flex',
          flexWrap: 'wrap',
          gap: 8,
        }}
      >
        <input
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          placeholder="Search reports..."
          style={{ padding: 8, minWidth: 220, borderRadius: 8, border: '1px solid #99a6b3' }}
        />

        <select
          value={status}
          onChange={(e) => setStatus(e.target.value)}
          style={{ padding: 8, borderRadius: 8, border: '1px solid #99a6b3' }}
        >
          <option>All</option>
          <option>Ready</option>
          <option>Draft</option>
          <option>Blocked</option>
        </select>

        <button onClick={() => setShowHelp((prev) => !prev)} style={{ padding: '8px 10px', borderRadius: 8, border: '1px solid #99a6b3' }}>
          {showHelp ? 'Hide Help' : 'Show Help'}
        </button>

        <button onClick={() => setPinReady((prev) => !prev)} style={{ padding: '8px 10px', borderRadius: 8, border: '1px solid #99a6b3' }}>
          {pinReady ? 'Unpin Ready' : 'Pin Ready'}
        </button>
      </div>

      {toast && (
        <div
          style={{
            marginBottom: 12,
            border: `1px solid ${isLight ? '#98b6d6' : '#5d83a9'}`,
            borderRadius: 10,
            padding: 10,
            background: isLight ? '#eaf3ff' : '#2f4b67',
          }}
        >
          {toast}
        </div>
      )}

      {showHelp && (
        <div
          style={{
            marginBottom: 12,
            border: `1px dashed ${isLight ? '#7d8d9f' : '#7a8ea3'}`,
            borderRadius: 10,
            padding: 12,
            background: isLight ? '#ffffff' : '#2b3541',
          }}
        >
          <strong>Help:</strong> Use search + status to narrow rows. "Open" loads a details panel and increments open count ({counter}).
        </div>
      )}

      <div
        style={{
          marginBottom: 12,
          border: `1px solid ${isLight ? '#c8d1db' : '#4b5b6d'}`,
          borderRadius: 10,
          padding: 12,
          background: isLight ? '#ffffff' : '#2b3541',
          display: 'flex',
          gap: 8,
          flexWrap: 'wrap',
        }}
      >
        <span style={{ color: badgeColor('Ready') }}>Ready: {readyCount}</span>
        <span style={{ color: badgeColor('Draft') }}>Draft: {draftCount}</span>
        <span style={{ color: badgeColor('Blocked') }}>Blocked: {blockedCount}</span>
      </div>

      {pinReady && (
        <div
          style={{
            marginBottom: 12,
            border: `1px solid ${isLight ? '#b7d8be' : '#55815d'}`,
            borderRadius: 10,
            padding: 12,
            background: isLight ? '#f2fff3' : '#2f4533',
          }}
        >
          <h3 style={{ marginTop: 0 }}>Pinned Ready Reports</h3>
          {pinnedReadyReports.map((report) => (
            <div key={report.id} style={{ padding: '6px 0', borderBottom: '1px solid rgba(120,120,120,0.25)' }}>
              {report.title} - {report.owner}
            </div>
          ))}
        </div>
      )}

      <div style={{ display: 'grid', gap: 8 }}>
        {filtered.map((report) => (
          <ReportRow key={report.id} report={report} theme={theme} onOpen={openReport} />
        ))}
      </div>

      {opened && <ReportRowDetails {...openDetails} />}
    </section>
  )
}
